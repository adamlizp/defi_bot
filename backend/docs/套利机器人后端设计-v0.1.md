### 功能范围

- 监控Uniswap和SushiSwap价格

## 🗄️ 数据库设计（MySQL）

### 3张表

#### 1. 套利机会表
```sql
CREATE TABLE opportunities (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    asset TEXT NOT NULL,
    amount_in TEXT NOT NULL,
    swap_path TEXT NOT NULL,        -- JSON数组
    dexes TEXT NOT NULL,            -- JSON数组
    expected_profit TEXT,
    status TEXT DEFAULT 'pending',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);
```

#### 2. 交易记录表
```sql
CREATE TABLE transactions (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    opportunity_id INTEGER,
    tx_hash TEXT,
    status TEXT,
    gas_used INTEGER,
    actual_profit TEXT,
    executed_at DATETIME DEFAULT CURRENT_TIMESTAMP
);
```

#### 3. 价格缓存表
```sql
CREATE TABLE price_cache (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    dex TEXT NOT NULL,
    token0 TEXT NOT NULL,
    token1 TEXT NOT NULL,
    price TEXT NOT NULL,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
);
```

## 💻 核心代码

### main.go

```go
package main

import (
    "log"
    "time"
    
    "github.com/defi-bot/backend/internal/monitor"
    "github.com/defi-bot/backend/internal/strategy"
    "github.com/defi-bot/backend/internal/executor"
)

func main() {
    log.Println("Starting DeFi Arbitrage Bot MVP...")
    
    // 加载配置
    cfg := LoadConfig("config.yaml")
    
    // 初始化组件
    blockchain := NewBlockchainClient(cfg.Blockchain.RpcURL)
    monitor := monitor.NewPriceMonitor(blockchain)
    strategy := strategy.NewTriangularStrategy(cfg.Arbitrage)
    executor := executor.NewExecutor(blockchain, cfg.Contracts)
    
    // 启动价格监控
    go monitor.Start()
    
    // 主循环
    ticker := time.NewTicker(time.Duration(cfg.Arbitrage.CheckIntervalSec) * time.Second)
    defer ticker.Stop()
    
    for range ticker.C {
        // 获取最新价格
        prices := monitor.GetLatestPrices()
        
        // 查找套利机会
        opp := strategy.FindOpportunity(prices)
        
        if opp != nil {
            log.Printf("Found opportunity! Profit: %s", opp.ExpectedProfit)
            
            // 执行套利
            if tx, err := executor.Execute(opp); err != nil {
                log.Printf("Execution failed: %v", err)
            } else {
                log.Printf("Transaction sent: %s", tx.Hash().Hex())
            }
        }
    }
}
```
