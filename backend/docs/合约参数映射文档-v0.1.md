## 📋 核心参数映射

### 1. ArbitrageParams (IArbitrage.sol)

#### 合约定义
```solidity
struct ArbitrageParams {
    address asset;           // 资产地址
    uint256 amountIn;        // 输入金额
    address[] swapPath;      // 交易路径
    address[] dexes;         // DEX地址数组
    uint256 minProfit;       // 最小利润
}
```

#### Go结构体
```go
type ArbitrageParams struct {
    Asset     common.Address   `json:"asset"`
    AmountIn  *big.Int         `json:"amountIn"`
    SwapPath  []common.Address `json:"swapPath"`
    Dexes     []common.Address `json:"dexes"`
    MinProfit *big.Int         `json:"minProfit"`
}
```

#### 数据库表（MySQL）
```sql
CREATE TABLE opportunities (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    asset VARCHAR(42) NOT NULL,        -- 对应asset
    amount_in VARCHAR(78) NOT NULL,    -- 对应amountIn
    swap_path JSON NOT NULL,           -- JSON数组对应swapPath
    dexes JSON NOT NULL,               -- JSON数组对应dexes  
    expected_profit VARCHAR(78),       -- 对应计算的profit
    status VARCHAR(20) DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```

## 类型转换

### Solidity → Go
```go
// address → common.Address
addr := common.HexToAddress("0xC02a...")

// uint256 → *big.Int
amount := big.NewInt(1000000000000000000) // 1 ETH in Wei

// address[] → []common.Address
path := []common.Address{
    common.HexToAddress("0x..."),
    common.HexToAddress("0x..."),
}
```

### Go → 数据库（MySQL）
```go
// *big.Int → VARCHAR (字符串存储)
amountStr := amount.String()
db.Exec("INSERT INTO ... VALUES (?)", amountStr)

// []common.Address → JSON
pathJSON, _ := json.Marshal(path)
db.Exec("INSERT INTO ... VALUES (?)", string(pathJSON))

// 使用GORM
db.Create(&Opportunity{
    Asset:     asset.Hex(),
    AmountIn:  amount.String(),
    SwapPath:  pathJSON,
    Dexes:     dexesJSON,
})
```



## 单位换算

### Wei / Ether
```go
// Ether → Wei
func EtherToWei(ether float64) *big.Int {
    wei := new(big.Float).Mul(
        big.NewFloat(ether),
        big.NewFloat(1e18),
    )
    result, _ := wei.Int(nil)
    return result
}

// Wei → Ether
func WeiToEther(wei *big.Int) float64 {
    ether := new(big.Float).SetInt(wei)
    ether.Quo(ether, big.NewFloat(1e18))
    result, _ := ether.Float64()
    return result
}
```

### Basis Points (bps)
```go
// 1 bps = 0.01%
// 100 bps = 1%
// 1000 bps = 10%

func BpsToPercent(bps uint16) float64 {
    return float64(bps) / 100.0
}
```

## 🔧 合约调用示例

### executeStrategy

```go
// 准备参数
params := contract.ArbitrageCoreArbitrageParams{
    Asset: common.HexToAddress("0xC02a..."), // WETH
    AmountIn: big.NewInt(100000000000000000), // 0.1 ETH
    SwapPath: []common.Address{
        common.HexToAddress("0xC02a..."), // WETH
        common.HexToAddress("0xdAC1..."), // USDT
        common.HexToAddress("0xC02a..."), // WETH
    },
    Dexes: []common.Address{
        common.HexToAddress("0x7a25..."), // Uniswap
        common.HexToAddress("0xd9e1..."), // SushiSwap
    },
    MinProfit: big.NewInt(5000000000000000), // 0.005 ETH
}

// 调用合约
tx, err := arbitrageCoreContract.ExecuteStrategy(
    opts,
    0, // OWN_FUNDS策略
    params,
)
```
